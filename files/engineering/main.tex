\cvrow{Experience}{%
  \workentry
    {\textbf{Graduate Student Researcher} @ Applied Cryptography Group, CQUPT}
    {Sep. 2024--Present}

  \workproject{\textbf{Private Information Retrieval for Hyperledger Fabric}}
  \workprojectdetail{Designed a private-read mechanism for the Hyperledger Fabric
  ledger world state based on a PIR protocol using the BGV HE scheme from the
  Lattigo library.}
  \workprojectdetail{Developed a threat model with an honest-but-curious (HBC) node
  and demonstrated query privacy based on the IND-CPA security of BGV, including
  repeated queries and observable information leakage.}

  \workproject{\textbf{Privacy-Preserving Aggregation for Federated Reinforcement Learning}}
  \workprojectdetail{Designed the TWAPPA-FRL architecture and implemented a private
  weighted aggregation mechanism for DQN updates based on the CKKS HE scheme using
  the TenSEAL library.}
  \workprojectdetail{Developed a threat model with an honest-but-curious (HBC)
  aggregator and demonstrated client-update privacy based on the IND-CPA security
  of CKKS, including observable information leakage.}

  \workproject{\textbf{Verifiable Multi-Keyword Searchable Encryption}}
  \workprojectdetail{Analyzed the threat model with an HBC data owner and user and
  a malicious cloud server; demonstrated query privacy in the real/ideal model
  based on OT security and the pseudorandomness of CI-rOPRF/PRG under explicitly
  defined search- and access-pattern leakage.}
  \workprojectdetail{Analyzed the justification of result verifiability using a
  file-key verification tag and AEAD, which detect key and file tampering by a
  malicious cloud server.}
}
