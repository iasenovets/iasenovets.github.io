\cvrow{Research\\Experience}{%
  \entry
    {\textbf{Graduate Student Researcher} @ Cryptography and Applications Research Group, CQUPT}
    {Sep. 2024--Present}
  \vspace{-0.85em}

  \vspace{0.35em}

  \textbf{Private Information Retrieval for Hyperledger Fabric}\par
  \begin{itemize}
    \item Designed and implemented a BGV-based computational PIR protocol in Go using Lattigo,
    integrating private world-state reads directly into Hyperledger Fabric chaincode.
    \item Defined the honest-but-curious (HBC) threat model and analysed query privacy under the
    IND-CPA security of BGV, including repeated-query security and observable leakage.
    \item Built the experimental workflow and evaluated correctness, latency, communication,
    storage, and deployment overhead across several cryptographic parameter sets.
  \end{itemize}

  \vspace{0.42em}

  \textbf{Privacy-Preserving Aggregation for Federated Reinforcement Learning}\par
  \begin{itemize}
    \item Co-developed TWAPPA-FRL, combining trust-weighted aggregation (TWA)
    with CKKS-based privacy-preserving aggregation (PPA) for collaborative
    intrusion detection, as an equal-contribution author.
    \item Combined trust-aware client weighting with CKKS-based encrypted aggregation of
    federated model updates and aggregate-only decryption by a separate non-training key authority.
    \item Developed the threat model and security analysis, including individual-update
    confidentiality, explicit leakage, and the assumptions and limitations of the PPA-enabled paths.
  \end{itemize}

  \vspace{0.42em}

  \textbf{Verifiable Multi-Keyword Searchable Encryption}\par
  \begin{itemize}
    \item Contributed to EVMK-SSE, an efficient multi-keyword searchable encryption
    scheme based on an OT-based client-independent relaxed OPRF.
    \item Reviewed the security analysis covering query privacy, explicit search and
    access-pattern leakage, and result verifiability against a malicious cloud server.
  \end{itemize}
}
